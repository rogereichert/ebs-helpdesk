5 - Estrutura recomendada para config/db.js
// config/db.js

const mysql = require('mysql2');

// Criação da conexão com o banco
const db = mysql.createConnection({
  host: process.env.DB_HOST,
  user: process.env.DB_USER,
  password: process.env.DB_PASS,
  database: process.env.DB_NAME
});

module.exports = db;

6 - server.js
// server.js

// 1. Carrega as variáveis de ambiente do .env
require('dotenv').config();

// 2. Importa os módulos necessários
const express = require('express');
const cors = require('cors');
const app = express();

// 3. Importa a conexão com o banco de dados
const db = require('./config/db');

// 4. Middlewares globais
app.use(cors()); // Permite requisições de diferentes origens
app.use(express.json()); // Converte o body das requisições para JSON

// 5. Testa conexão com o banco ao iniciar
db.connect((err) => {
  if (err) {
    console.error('Erro ao conectar ao banco de dados:', err);
    process.exit(1); // Encerra o servidor se não conectar
  }
  console.log('Conectado ao banco de dados MySQL!');
});

// 6. Rotas principais (iremos separar depois em arquivos próprios)
app.get('/', (req, res) => {
  res.send('API de chamados iniciada com sucesso!');
});

// 7. Porta e start do servidor
const PORT = process.env.PORT || 3000;
app.listen(PORT, () => {
  console.log(`Servidor rodando em http://localhost:${PORT}`);
});
